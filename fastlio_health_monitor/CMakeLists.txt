# 独立包构建顺序：接口 -> 源码清单 -> C++核心 -> ROS节点 -> 可选RViz -> 可选测试。
cmake_minimum_required(VERSION 3.0.2)
project(fastlio_health_monitor)
add_compile_options(-std=c++14 -Wall -Wextra)
# 无界面机器人设OFF，只编译核心与三个C++节点。
option(BUILD_HEALTH_RVIZ_PANEL "Build the optional RViz/Qt health panel" ON)
# 标准PointCloud2不需要Livox；只有接CustomMsg时才设ON并提供已安装的驱动。
option(BUILD_HEALTH_LIVOX_ADAPTER "Build Livox CustomMsg adapter" OFF)
set(HEALTH_ROS_DEPS diagnostic_msgs geometry_msgs message_generation nav_msgs roscpp rosbag rosgraph_msgs sensor_msgs std_msgs std_srvs)
set(HEALTH_PANEL_DEPS)
set(HEALTH_LIBRARIES robot_health_core)
if(BUILD_HEALTH_RVIZ_PANEL)
  list(APPEND HEALTH_PANEL_DEPS pluginlib rviz)
  list(APPEND HEALTH_LIBRARIES fastlio_health_monitor_panel)
endif()
find_package(catkin REQUIRED COMPONENTS ${HEALTH_ROS_DEPS} ${HEALTH_PANEL_DEPS})
# 内部控制消息保留实例/代际协议；日志字段精简不破坏消息MD5兼容性。
add_message_files(FILES RuntimeState.msg TaskState.msg)
add_service_files(FILES ControlModule.srv)
generate_messages(DEPENDENCIES std_msgs)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Boost REQUIRED COMPONENTS filesystem system)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
catkin_package(INCLUDE_DIRS include ${EIGEN3_INCLUDE_DIRS}
  LIBRARIES ${HEALTH_LIBRARIES}
  CATKIN_DEPENDS diagnostic_msgs geometry_msgs message_runtime nav_msgs roscpp rosbag rosgraph_msgs sensor_msgs std_msgs std_srvs ${HEALTH_PANEL_DEPS})
include_directories(include ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
# 构建材料清单由简单Python脚本生成，实际健康计算全部为C++。
add_custom_target(health_source_manifest ALL
  BYPRODUCTS ${CMAKE_CURRENT_BINARY_DIR}/source_build_manifest.h ${CMAKE_CURRENT_BINARY_DIR}/source_build_manifest.json
  COMMAND /usr/bin/python3 ${CMAKE_CURRENT_SOURCE_DIR}/scripts/build_manifest.py ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/source_build_manifest.json)
add_library(robot_health_core src/health_core.cpp src/event_recorder.cpp)
add_dependencies(robot_health_core health_source_manifest)
target_include_directories(robot_health_core PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(robot_health_core yaml-cpp ${Boost_LIBRARIES} OpenSSL::Crypto ${CMAKE_THREAD_LIBS_INIT} ${CMAKE_DL_LIBS})
# ROS接入层、受管播放器和进程管理器均复用同一个核心库。
add_executable(robot_health_monitor_node src/robot_health_monitor_node.cpp src/input_adapters.cpp)
add_executable(health_bag_playback_node src/bag_playback_node.cpp)
add_executable(robot_runtime_controller_node src/runtime_controller_node.cpp)
foreach(target robot_health_monitor_node health_bag_playback_node robot_runtime_controller_node)
  add_dependencies(${target} ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
  target_link_libraries(${target} robot_health_core ${catkin_LIBRARIES})
endforeach()
if(BUILD_HEALTH_LIVOX_ADAPTER)
  find_package(livox_ros_driver REQUIRED)
  target_compile_definitions(robot_health_monitor_node PRIVATE HEALTH_WITH_LIVOX)
  target_include_directories(robot_health_monitor_node PRIVATE ${livox_ros_driver_INCLUDE_DIRS})
  target_link_libraries(robot_health_monitor_node ${livox_ros_driver_LIBRARIES})
  add_dependencies(robot_health_monitor_node ${livox_ros_driver_EXPORTED_TARGETS})
endif()
# GUI不是机器人的运行前提；只在选择ON时查找Qt和RViz依赖。
if(BUILD_HEALTH_RVIZ_PANEL)
  find_package(Qt5 REQUIRED COMPONENTS Widgets)
  qt5_wrap_cpp(HEALTH_PANEL_MOC include/fastlio_health_monitor/health_panel.h)
  add_library(fastlio_health_monitor_panel src/health_panel.cpp ${HEALTH_PANEL_MOC})
  add_dependencies(fastlio_health_monitor_panel ${catkin_EXPORTED_TARGETS})
  target_link_libraries(fastlio_health_monitor_panel ${catkin_LIBRARIES} Qt5::Widgets)
endif()
# 测试只在明确开启CATKIN_ENABLE_TESTING时构建，不混入正常机器人运行过程。
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(health_core_test test/health_core_test.cpp test/optional_rules_test.cpp)
  if(TARGET health_core_test)
    target_link_libraries(health_core_test robot_health_core)
    target_compile_definitions(health_core_test PRIVATE HEALTH_TEST_CONFIG_DIR="${CMAKE_CURRENT_SOURCE_DIR}/config")
  endif()
endif()
# install后保留配置、接口、launch、RViz资源及可读头文件。
install(TARGETS robot_health_monitor_node health_bag_playback_node robot_runtime_controller_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(TARGETS ${HEALTH_LIBRARIES} ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY launch config rviz DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES health_panel_plugin.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(PROGRAMS scripts/start_localization.sh DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
